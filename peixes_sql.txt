criando os fornecedores:
create table fornecedores(
	id_fornecedor int not null auto_increment primary key,
    nome varchar(100) not null,
    telefone varchar(20) not null,
    cidade varchar(60) not null,
    produto_fornecido varchar(80) not null
);

criando os tanques:
create table tanques (
	id_tanque int primary key not null auto_increment,
    identificacao varchar(30) not null,
    capacidade int not null,
    localizacao varchar(80) not null,
    status_tanque varchar(30) not null
);

criando os produção:
create table producao (
	id_producao int primary key not null auto_increment,
    id_tanque int not null,
    especie varchar(60) not null unique,
    qtd_peixes int not null,
    peso_medio decimal(10, 2) not null,
    data_registro date not null,
    foreign key(id_tanque)
	references tanques(id_tanque)
);

criando os clientes:
create table clientes (
	id_cliente int primary key not null auto_increment,
    nome varchar(100) not null,
    telefone varchar(20) not null,
    cidade varchar(60) not null,
    tipo_cliente varchar(40) not null
);

criando os vendas:
create table vendas (
	id_venda int primary key not null auto_increment,
    id_cliente int not null,
    especie varchar(60) not null,
    qtd_kg decimal(10, 2) not null,
    preco_kg decimal(10, 2) not null,
    valor_total decimal(10, 2) not null,
    data_venda date not null,
    foreign key(id_cliente) references clientes(id_cliente),
    constraint fk_vendas_producao foreign key (especie) references producao (especie)
);
